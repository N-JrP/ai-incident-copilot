async function wait(ms){ return new Promise(r => setTimeout(r, ms)); }

async function loadAllStartups() {
  let previousCount = 0;
  let stuckRounds = 0;

  for (let i = 0; i < 120; i++) {
    const buttons = [...document.querySelectorAll("button")];

    const loadMore = buttons.find(b => {
      const text = (b.innerText || "").trim().toLowerCase();
      return text.startsWith("mehr laden");
    });

    if (loadMore) {
      loadMore.scrollIntoView({behavior: "smooth", block: "center"});
      await wait(800);
      loadMore.click();
      await wait(2500);
    } else {
      window.scrollTo(0, document.body.scrollHeight);
      await wait(1500);
    }

    const cards = [...document.querySelectorAll("a")]
      .map(a => ({
        name: (a.innerText || "").trim().replace(/\s+/g, " "),
        url: a.href
      }))
      .filter(x =>
        x.name &&
        x.url &&
        !x.url.includes("#") &&
        !x.url.includes("datenschutz") &&
        !x.url.includes("impressum") &&
        !x.url.includes("kontakt")
      );

    console.log("Current links found:", cards.length);

    if (cards.length === previousCount) stuckRounds++;
    else stuckRounds = 0;

    previousCount = cards.length;

    if (stuckRounds >= 5) break;
  }

  const rows = [...document.querySelectorAll("a")]
    .map(a => ({
      name: (a.innerText || "").trim().replace(/\s+/g, " "),
      url: a.href
    }))
    .filter(x =>
      x.name &&
      x.url &&
      !x.url.includes("#") &&
      !x.url.includes("datenschutz") &&
      !x.url.includes("impressum") &&
      !x.url.includes("kontakt") &&
      !x.url.includes("startplatz.de/en") &&
      !x.url.includes("twitter") &&
      !x.url.includes("facebook") &&
      !x.url.includes("instagram")
    );

  const unique = Array.from(new Map(rows.map(x => [x.name + x.url, x])).values());

  const csv = "Startup Name,Website/URL\n" +
    unique.map(r => `"${r.name.replaceAll('"','""')}","${r.url}"`).join("\n");

  const blob = new Blob([csv], {type: "text/csv"});
  const url = URL.createObjectURL(blob);
  const a = document.createElement("a");
  a.href = url;
  a.download = "startplatz_startups.csv";
  a.click();

  console.log("DONE. CSV downloaded. Total rows:", unique.length);
}

loadAllStartups();
